function(add_unit_test name source)
    add_executable(${name} ${source})
    target_link_libraries(${name} PRIVATE unity)
    target_compile_options(${name} PRIVATE --coverage)
    target_link_options(${name} PRIVATE --coverage)
    add_test(NAME ${name} COMMAND ${name})
endfunction()

add_unit_test(test_range test_range.c)

target_link_libraries(test_ascii PRIVATE unity)
target_compile_options(test_ascii PRIVATE --coverage)
target_link_options(test_ascii PRIVATE --coverage)
